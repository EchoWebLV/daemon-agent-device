# "main" component — the ESP-IDF entry point. PlatformIO auto-generates the
# outer CMakeLists.txt and expects this file here.
#
# ESP-IDF auto-wires the common "main" dependency chain (log, freertos,
# esp_system, heap, nvs_flash, …) so we don't need to list those explicitly.
# Add to REQUIRES only when we depend on something outside that set.
idf_component_register(
    SRCS
        "ai.c"
        "app_main.c"
        "base58.c"
        "bus.c"
        "buttons.c"
        "config_screen.c"
        "creature_screen.c"
        "creatures_data.c"
        "devcfg.c"
        "display.c"
        "info_screen.c"
        "mic.c"
        "menu_screen.c"
        "price.c"
        "server.c"
        "social_x.c"
        "settings_screen.c"
        "solana_tx.c"
        "stt.c"
        "swap.c"
        "swap_screen.c"
        "testharness.c"
        "touch.c"
        "ui.c"
        "voice.c"
        "wallet.c"
        "wallet_screen.c"
        "wifi_screen.c"
        "wifi_sta.c"
        "x_post_screen.c"
        "x402.c"
        # Generated by scripts/embed_assets.py (wired in via platformio.ini's
        # extra_scripts). Exposes index_html_start[] + index_html_len, which
        # server.c serves for GET /.
        "generated/index_html.c"
        "generated/mcp_mjs.c"
    INCLUDE_DIRS
        "."
    REQUIRES
        # ST7789 bring-up over SPI; CST328 bring-up over I2C master.
        esp_lcd
        driver
        esp_driver_i2c
        # Voice: I2S_STD driver for PCM5101 DAC output.
        esp_driver_i2s
        # NVS-backed settings + Wi-Fi STA bring-up + HTTP server.
        nvs_flash
        esp_wifi
        esp_netif
        esp_event
        esp_http_server
        # Outbound HTTP/HTTPS + root CA bundle (for Helius / CoinGecko /
        # Gemini in phase 4). cJSON lives in the `json` component.
        esp_http_client
        esp-tls
        json
        # Wallet signing (Ed25519) for x402. orlp/ed25519 is vendored under
        # /components/ed25519 and used only by wallet.c / solana_tx.c.
        ed25519
        # mbedtls for SHA-256 (x402 facilitator digest) + base64 + TLS bits.
        mbedtls
        # esp_timer for wallet refresh age tracking.
        esp_timer
        # LVGL + the IDF port layer that glues it to esp_lcd.
        # Pulled in via src/idf_component.yml.
        lvgl
        esp_lvgl_port
        # LCD panel drivers — ILI9342 (BOX-3) over esp_lcd, plus the
        # generic touch abstraction and the GT911 vendor driver.
        esp_lcd_ili9341
        esp_lcd_touch
        esp_lcd_touch_gt911
        # Audio codec stack — ES8311 for the speaker out path on BOX-3.
        # Pulls in i2s_data_if + i2c_ctrl_if + es8311_codec as one component.
        esp_codec_dev
        # BOX-3 BSP — audio init helpers (bsp_audio_codec_*_init etc.).
        # We bypass its display/touch helpers by simply not calling them.
        esp-box-3
        # iot_button — front-face PTT button via the GT911's key bit.
        # See buttons.c. Already pulled transitively through esp-box-3,
        # but listed explicitly because we use its API directly.
        button
        # mDNS — advertise `daemon.local` once Wi-Fi associates so the MCP
        # server (and curl from any laptop) can reach the device by name.
        mdns
)
